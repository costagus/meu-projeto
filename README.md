# meu

git init
- iniciar novo projeto com git

git add <nome-arquivi> ou .
- add os arquivos prontos para serem commitados

git commit -m "mensagem"
- commit arquivos no historico

git log
- mostra os ultimos commit, log de alteracoes

git status 
- estado das ramificacoes

git diff
- oq foi alterado
- o que tem de alteracao

git merge
- merge de ramificacoes, mescla ramificacoes

git branch
- mostra branch atual

git checkout <nomedabranch>
- muda p/ essa branch

git branch -b <nomedabranch>
- cria uma nova branch a partir de uma atual

git remote add <nome> <url>
= add um novo repositorio remoto

git push <nome> <noemdabranc>
- manda noosas alteracoes locais para o repositorio rmoto para cada branch

git pull 
- pega as alteracoes do remoto e jgoa para nossa maquina

git fetch
- atualiza o historico local de acordo com o remoto
- sincronicazao do local c remoto
